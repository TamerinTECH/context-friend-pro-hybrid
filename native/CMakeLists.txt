cmake_minimum_required(VERSION 3.22)
project(ContextFriendNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory(../.deps/llama llama)
add_subdirectory(../.deps/whisper whisper)
if(ANDROID)
  add_library(cf_native SHARED engine.cpp android.cpp)
  target_link_libraries(cf_native PRIVATE llama whisper log)
  target_link_options(cf_native PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_library(cf_native STATIC engine.cpp)
  target_link_libraries(cf_native PRIVATE llama whisper)
  if(APPLE)
    target_sources(cf_native PRIVATE speakers.cpp)
    target_include_directories(cf_native PRIVATE ../.deps/sherpa)
  endif()
endif()
target_include_directories(cf_native PUBLIC include)
if(NOT ANDROID AND NOT APPLE)
  add_library(cf_speaker_api_check OBJECT speakers.cpp)
  target_include_directories(cf_speaker_api_check PRIVATE include ../.deps/sherpa)
  add_executable(cf_probe probe.cpp)
  target_link_libraries(cf_probe PRIVATE cf_native)
endif()
